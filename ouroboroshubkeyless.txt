-- Kiểm tra xem người chơi đã nhập đúng key hay chưa
local correctKey = "OuroborosHubKeyHsjskwjKalwpwoMamqoekeo"
local TweenService = game:GetService("TweenService")

-- Tên file lưu trạng thái key (Lưu 24 giờ)
local saveFileName = "OuroborosHub_KeyData.json"
local dayInSeconds = 86400 -- 24 giờ tính bằng giây

-- Hàm kiểm tra xem key đã được lưu và còn hạn hay không
local function hasValidKey()
    if readfile and writefile and isfile then
        if isfile(saveFileName) then
            local success, data = pcall(function()
                return game:GetService("HttpService"):JSONDecode(readfile(saveFileName))
            end)
            if success and data and data.key == correctKey and (os.time() - data.time < dayInSeconds) then
                return true
            end
        end
    end
    return false
end

-- Hàm lưu key khi nhập đúng
local function saveKey()
    if writefile and game:GetService("HttpService") then
        local data = {
            key = correctKey,
            time = os.time()
        }
        pcall(function()
            writefile(saveFileName, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Hàm chạy script chính
local function loadMainScript()
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/joustingmatch/Ouroboros/main/loader.lua"))()
    end)
    if not success then
        warn("Error loading script: " .. tostring(err))
    end
end

-- Nếu đã lưu key hợp lệ trong vòng 24h thì bỏ qua bảng Key System luôn
if hasValidKey() then
    loadMainScript()
    return
end

-- Hiệu ứng lắc siêu mượt
local function smoothShake()
    local originalPos = MainFrame.Position
    local tweenInfo = TweenInfo.new(0.05, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    local shakeOffsets = {-8, 8, -6, 6, -3, 3, 0}
    
    for _, offset in ipairs(shakeOffsets) do
        local targetPos = originalPos + UDim2.new(0, offset, 0, 0)
        local tween = TweenService:Create(MainFrame, tweenInfo, {Position = targetPos})
        tween:Play()
        tween.Completed:Wait()
    end
    
    MainFrame.Position = originalPos
end

        loadMainScript()
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        StatusLabel.Text = "Incorrect key! Please try again."
        TextBox.Text = ""
        smoothShake()
    end
end)
